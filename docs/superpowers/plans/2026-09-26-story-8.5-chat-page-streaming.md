## Review Focus

Five failure modes the ACs imply but no default step's test would catch — each gets a pinning test added to its owning step below:

1. **The Thinking Indicator must render synchronously on submit, before any network activity runs** (NFR-Latency) — pinned in Task 4 (`chatStore.test.ts`, asserts the optimistic state update happens before `streamChat`'s promise ever resolves) and Task 9 (`Chat.test.tsx`, the same assertion at the page level).
2. **Exactly one terminal SSE event, regardless of how `chatTurn` fails** (a thrown error, a `Result` failure, or zero `emit` calls at all) — pinned in Task 1's `runChatStream` tests (three separate cases).
3. **A broken/disconnected stream must never throw out of the request handler** (`writeSSE` rejecting, simulating a client that vanished mid-stream) — pinned in Task 1.
4. **SSE parsing must survive a chunk boundary landing mid-line**, not only between blocks (a slow/fragmented TCP read can split a `data:` line itself) — pinned in Task 3's `parseSseStream` test.
5. **The "chat not configured" path must still produce a stream the client's own parser can consume** (not a bare JSON body incompatible with SSE framing) — pinned server-side in Task 1 and client-side in Task 3 (feeding that literal single-error-event shape into `parseSseStream`).

```ts
import { chatTurn, type ChatTurnDeps } from "../app/chat-turn.ts";
import type { ChatSession } from "../app/chat-session.ts";
import type { ChatStreamEvent, ChatTurnRequest } from "../types/api.ts";
import { createAnthropicMessagesClient, loadLlmAdapterConfigFromEnv } from "../adapters/llm-adapter.ts";

// ============================================================================
// POST /api/chat — its own SSE response (Story 8.5, AD-18, contract C5).
// ============================================================================

/** The slice of Hono's `SSEStreamingApi` this route needs — narrower than `SseStreamLike` (no `aborted`/`onAbort`: a disconnect just makes `writeSSE` a no-op, per Hono's own `write()`, which swallows write errors after the client is gone). */
export interface ChatSseStreamLike {
  writeSSE(message: { readonly event: string; readonly data: string }): Promise<void>;
}

/** `chatTurn`'s exact contract signature (C3) — injectable so this file's own tests never depend on Task 4's real `ChatTurnDeps` field set, only on this shape. Defaults to the real `chatTurn`. */
export type ChatTurnFn = (deps: ChatTurnDeps, input: ChatTurnRequest) => Promise<Result<import("../types/domain.ts").ChatTurnResponse extends never ? never : import("../types/api.ts").ChatTurnResponse, import("../types/domain.ts").YohError>>;

function serializeEvent(event: ChatStreamEvent): { event: string; data: string } {
  return { event: event.type, data: JSON.stringify(event) };
}

/**
 * Drives one `POST /api/chat` SSE response. `chatTurn` pushes zero or more
 * `status`/`delta` events through `emit` as it runs — this function alone
 * decides the terminal framing, converting `chatTurn`'s settled `Result`
 * into exactly one `done`/`error` event afterwards (Assumption 2: `chatTurn`
 * itself never emits one). A thrown error (a bug, not a `Result` failure)
 * is caught here too, so the stream never hangs and Hono's own `streamSSE`
 * never sees the exception — which would otherwise write its own bare-
 * string `error` event, shaped differently from `ChatStreamEvent`. Because
 * `stream.writeSSE`'s underlying `WritableStreamDefaultWriter` processes
 * writes strictly in call order, awaiting only the FINAL write (below)
 * guarantees every earlier, unawaited `emit`-triggered write has already
 * been processed first — no explicit flush needed.
 */
export async function runChatStream(
  stream: ChatSseStreamLike,
  deps: ChatTurnDeps,
  input: ChatTurnRequest,
  runChatTurn: ChatTurnFn = chatTurn,
): Promise<void> {
  const emit = (event: ChatStreamEvent): void => {
    void stream.writeSSE(serializeEvent(event));
  };
  try {
    const result = await runChatTurn({ ...deps, emit }, input);
    const terminal: ChatStreamEvent = result.ok ? { type: "done", response: result.value } : { type: "error", error: result.error };
    await stream.writeSSE(serializeEvent(terminal));
  } catch (err) {
    const message = err instanceof Error ? err.message : String(err);
    await stream.writeSSE(serializeEvent({ type: "error", error: { kind: "unreachable", message } }));
  }
}
```

```ts
/**
 * web/src/lib/chatStream.ts
 *
 * Story 8.5, AD-18/contract C6: `fetch` POST `/api/chat` and parse its
 * `text/event-stream` body into `ChatStreamEvent`s — there is no
 * `EventSource` for POST, so this is its own small SSE-over-fetch parser
 * rather than a second `connectEventStream` (`eventBus.ts` stays the one
 * `EventSource`, used only for hints). Uses the typed Hono RPC client for
 * the request itself (AD-17's "the ONE way web/ talks to the server"),
 * then reads `.body` directly instead of `.json()`.
 */
import { apiClient } from "./apiClient.ts";
import type { ChatStreamEvent, ChatTurnRequest } from "../../../src/types/api.ts";

export interface StreamChatHandlers {
  onEvent(event: ChatStreamEvent): void;
}

/**
 * Parses one `text/event-stream` byte stream into `ChatStreamEvent`s,
 * calling `onEvent` for every complete `event:`/`data:` block — blocks are
 * separated by a blank line, per the wire format `hono/streaming`'s
 * `writeSSE` writes (`shell/server.ts`'s `runChatStream`). Buffers across
 * `read()` calls so a chunk boundary landing mid-block (or mid-line) never
 * loses or corrupts a block. A block with no `event:` line is ignored (a
 * bare keep-alive comment, never actually emitted by this route, but
 * tolerated defensively — the same posture `events.ts` takes toward
 * `GET /api/events`'s keep-alive comments).
 */
export async function parseSseStream(body: ReadableStream<Uint8Array>, onEvent: (event: ChatStreamEvent) => void): Promise<void> {
  const reader = body.getReader();
  const decoder = new TextDecoder();
  let buffer = "";
  for (;;) {
    const { done, value } = await reader.read();
    if (done) break;
    buffer += decoder.decode(value, { stream: true });
    let sepIndex = buffer.indexOf("\n\n");
    while (sepIndex !== -1) {
      const block = buffer.slice(0, sepIndex);
      buffer = buffer.slice(sepIndex + 2);
      let eventType: string | undefined;
      const dataLines: string[] = [];
      for (const line of block.split("\n")) {
        if (line.startsWith("event:")) eventType = line.slice(6).trim();
        else if (line.startsWith("data:")) dataLines.push(line.slice(5).trim());
      }
      if (eventType !== undefined && dataLines.length > 0) {
        onEvent(JSON.parse(dataLines.join("\n")) as ChatStreamEvent);
      }
      sepIndex = buffer.indexOf("\n\n");
    }
  }
}

/**
 * Sends one chat turn and streams its `ChatStreamEvent`s to
 * `handlers.onEvent` as they arrive. Resolves once the stream ends (after
 * its terminal `done`/`error` event) — never rejects for a mid-stream
 * `error` event (a normal, handled outcome); it rejects only for a genuine
 * transport failure (the `fetch` itself failing), which `chatStore.ts`
 * (Task 4) renders as its own local failure, the same way a check-off
 * write failure does (`notifications.ts`'s `addLocalFailureNotice`).
 */
export async function streamChat(request: ChatTurnRequest, handlers: StreamChatHandlers): Promise<void> {
  const res = await apiClient.api.chat.$post({ json: request });
  if (!res.body) throw new Error("chatStream: response had no body");
  await parseSseStream(res.body, handlers.onEvent);
}
```

```ts
/**
 * web/src/lib/chatStore.ts
 *
 * Story 8.5, contract C6, AD-10/AD-17 (`[DECISION DEFAULT]`, spine
 * Deferred/UX OQ13): the ONE chat transcript + unsent-draft store —
 * module-level state via `useSyncExternalStore` (same convention as
 * `homeView.ts`/`notifications.ts`), client memory only, never persisted.
 * Survives a page swipe or the Screensaver because `PageShell.tsx` keeps
 * every page mounted. Shared by the Chat page (this story) and the Home
 * Chat Bubble (Story 8.8).

import { useSyncExternalStore } from "react";
import { streamChat } from "./chatStream.ts";
import { addLocalFailureNotice } from "./notifications.ts";
import type { ChatStreamEvent, ChatTurnRequest } from "../../../src/types/api.ts";

export interface ChatViewMessage {
  readonly id: string;
  readonly role: "user" | "assistant";
  readonly text: string;
  readonly receipts: readonly string[];
  /** Plain text for now — Story 8.6 upgrades this to a Structured Question. */
  readonly question?: string;
  readonly status: "done" | "streaming" | "error";
  /** The latest `status` event's text — only meaningful while `status === "streaming"`. */
  readonly statusText?: string;
}

export interface ChatStoreState {
  readonly messages: readonly ChatViewMessage[];
  readonly draft: string;
  readonly sending: boolean;
}

let state: ChatStoreState = { messages: [], draft: "", sending: false };
const listeners = new Set<() => void>();

function set(next: ChatStoreState): void {
  state = next;
  listeners.forEach((l) => l());
}
function snapshot(): ChatStoreState {
  return state;
}

export function useChatStore(): ChatStoreState {
  return useSyncExternalStore(
    (onChange) => {
      listeners.add(onChange);
      return () => listeners.delete(onChange);
    },
    snapshot,
  );
}

export function setDraft(draft: string): void {
  set({ ...state, draft });
}

function historyFor(messages: readonly ChatViewMessage[]): ChatTurnRequest["history"] {
  return messages.filter((m) => m.status !== "streaming").map((m) => ({ role: m.role, content: m.text }));
}

function updateAssistant(id: string, patch: Partial<ChatViewMessage>): void {
  set({ ...state, messages: state.messages.map((m) => (m.id === id ? { ...m, ...patch } : m)) });
}

/**
 * Sends `message` as Spencer's turn. His turn (and a placeholder for
 * Yoh's, showing the Thinking Indicator) appear immediately, before any
 * network activity (NFR-Latency) — the draft clears in the same
 * synchronous update. A mid-stream `error` event (`chatTurn` ran but
 * failed, e.g. an upstream outage) marks the placeholder `"error"` without
 * discarding any text already streamed — the exchange happened, so it
 * stays in the transcript (`ChatMessage.tsx` renders a small "couldn't
 * finish" caption alongside it). A genuine transport failure (the request
 * never meaningfully completed) instead removes both placeholders,
 * restores the draft, and raises a local failure notice (`notifications.ts`'s
 * established `addLocalFailureNotice`, Story 7.10) — "no silent failure"
 * (EXPERIENCE.md).
 */
export async function send(message: string): Promise<void> {
  const trimmed = message.trim();
  if (trimmed === "" || state.sending) return;

  const userId = crypto.randomUUID();
  const assistantId = crypto.randomUUID();
  const historyBefore = historyFor(state.messages);
  const userMessage: ChatViewMessage = { id: userId, role: "user", text: trimmed, receipts: [], status: "done" };
  const assistantMessage: ChatViewMessage = { id: assistantId, role: "assistant", text: "", receipts: [], status: "streaming", statusText: "Thinking…" };
  set({ messages: [...state.messages, userMessage, assistantMessage], draft: "", sending: true });

  const request: ChatTurnRequest = { message: trimmed, history: [...historyBefore, { role: "user", content: trimmed }] };

  const onEvent = (event: ChatStreamEvent): void => {
    if (event.type === "status") {
      updateAssistant(assistantId, { statusText: event.text });
    } else if (event.type === "delta") {
      const current = state.messages.find((m) => m.id === assistantId);
      updateAssistant(assistantId, { text: (current?.text ?? "") + event.text });
    } else if (event.type === "done") {
      updateAssistant(assistantId, {
        text: event.response.reply,
        receipts: event.response.receipts,
        question: event.response.question?.text,
        status: "done",
      });
      set({ ...state, sending: false });
    } else {
      updateAssistant(assistantId, { status: "error" });
      set({ ...state, sending: false });
    }
  };

  try {
    await streamChat(request, { onEvent });
  } catch {
    set({
      messages: state.messages.filter((m) => m.id !== userId && m.id !== assistantId),
      draft: trimmed,
      sending: false,
    });
    addLocalFailureNotice("Couldn't reach Yoh — try again.");
  }
}

/** Test-only: clears module-level singleton state between tests. Never called from production code. */
export function __resetChatStoreForTests(): void {
  state = { messages: [], draft: "", sending: false };
}
```

```css
  opacity: 0.55;
  pointer-events: none;
```

```tsx
/**
 * web/src/components/ChatInput.tsx
 *
 * Story 8.5, UX-DR36, DESIGN.md `chat-input`: glass, always wide, the same
 * material/focus treatment as the Chat Bubble — reuses `.notification-glass`
 * (Story 7.7's shared floating-glass mechanism; DESIGN.md's Glass row lists
 * Chat Input under the same one mechanism as the Chat Bubble/Undo
 * Toast/Command Palette). Enter sends; Shift+Enter inserts a newline. The
 * draft is `chatStore.ts`'s own `draft` field — the ONE source of truth, so
 * it survives a page swipe or the Screensaver even if this component itself
 * remounts.
 */
import { useRef } from "react";
import { useChatStore, setDraft, send } from "../lib/chatStore.ts";

export function ChatInput(): React.JSX.Element {
  const { draft, sending } = useChatStore();
  const ref = useRef<HTMLTextAreaElement>(null);

  const submit = (): void => {
    if (draft.trim() === "") return;
    void send(draft);
    ref.current?.focus();
  };

  return (
    <div className="notification-glass sticky bottom-0 z-10 flex w-full items-end gap-2 rounded-full px-4 py-2 focus-within:shadow-[0_0_0_var(--focus-ring-width)_var(--color-accent-solid),0_0_16px_var(--color-accent-glow)]">
      <textarea
        ref={ref}
        rows={1}
        value={draft}
        placeholder="Ask Yoh, or type / for commands"
        aria-label="Message Yoh"
        className="max-h-32 flex-1 resize-none bg-transparent font-body text-body text-ink-primary outline-none placeholder:text-ink-secondary"
        onChange={(e) => setDraft(e.target.value)}
        onKeyDown={(e) => {
          if (e.key === "Enter" && !e.shiftKey) {
            e.preventDefault();
            submit();
          }
        }}
      />
      <button
        type="button"
        aria-label="Send"
        disabled={draft.trim() === "" || sending}
        onClick={submit}
        className="shrink-0 rounded-full px-3 py-1 font-body text-body text-accent-solid disabled:text-ink-secondary disabled:opacity-50"
      >
        Send
      </button>
    </div>
  );
}
```
